\begin{definition}[Product H{\"o}lder spaces]
  \label{def:product-holder-spaces}Let \(d_1, d_2 \geq 1\) and \(\alpha_1,
  \alpha_2 \in (0, 1)\) be two indexes define the homogeneous product
  H{\"o}lder seminorm by
  \begin{equation}
    \label{eq:holder-homogeneous-seminorm} \|f\|_{\dot{\mathcal{C}}^{\alpha_1,
    \alpha_2} (\R^{d_1} \times \R^{d_2})} \eqd \sup_{x \neq x'}  \frac{|f (x',
    y') + f (x, y) - f (x', y) - f (x, y') |}{|x' - x|^{\alpha_1} |y' -
    y|^{\alpha_2}} .
  \end{equation}
  Define the nonhomogeneous H{\"o}lder norm by
  \begin{equation}
    \label{eq:holder-nonhomogeneous-norm}
    \label{eq:holder-homogeneous-seminorm} \|f\|_{\mathcal{C}^{\alpha_1,
    \alpha_2} (\R^{d_1} \times \R^{d_2})} \eqd \begin{aligned}[t]
      &\|f\|_{L^{\infty}} + \sup_{\substack{
        x \neq x'\\
        |x' - x| < 1
      }} \sup_{\substack{
        y
      }}  \frac{|f (x', y) + f (x, y) |}{|x' - x|^{\alpha_1}}\\
      &{+ \sup_{\substack{
        x
      }}}  \sup_{\substack{
        y \neq y'\\
        |y' - y| < 1
      }} \frac{|f (x, y') + f (x, y) |}{|y' - y|^{\alpha_2}}\\
      &+ \sup_{\substack{
        x \neq x'\\
        |x' - x| < 1
      }} \sup_{\substack{
        y \neq y'\\
        |y' - y| < 1
      }} \frac{|f (x', y') + f (x, y) - f (x', y) - f (x, y')
      |}{|x' - x|^{\alpha_1} |y' - y|^{\alpha_2}}
    \end{aligned}
  \end{equation}
\end{definition}

\begin{definition}[Product wavelet coefficients]
  \label{def:product-wavelet-coefficients}Let \(d_1, d_2 \geq 1\) and \(\alpha_1,
  \alpha_2 \in (0, 1)\). For a function \(f\) of moderate growth and \(\varphi_j
  \in \Sch (\R^{d_j})\), \(j \in \{ 1, 2 \}\), define
  \begin{equation}
    \label{eq:wavelet-coefficient-fixed-test} \Em_{\varphi_1, \varphi_2} [f]
    (x, t_1, y, t_2) \eqd \int_{\R^{{d_1} }} \int_{\R^{{d_2} }} f (x', y')
    \Tr_x \Dil_{t_1} \varphi (x') \Tr_y \Dil_{t_2} \varphi (x')
    \hspace{0.17em} \dd x' \dd y'
  \end{equation}
  whenever the integral is absolutely convergent. We allow the test functions
  \(\varphi_j\), \(j \in \{ 1, 2 \}\) to depend on \((x, t_1, y, t_2)\) with
  straigh-forward modification of the definition (see
  \Cref{def:wavelet-coefficients})
\end{definition}
