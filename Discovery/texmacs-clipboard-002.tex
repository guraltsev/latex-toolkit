In this section we record the conventions and estimates for test functions,
wavelet coefficients, reconstruction operators, and convolution multipliers
used later. We write
\[ \R^{d + 1}_+ \eqd \R^d \times (0, \infty) . \]
We fix constants $0 < c_0 < C_0 < \infty$ and write $\1_{t \approx 1}$ for the
indicator of the interval $[c_0, C_0]$. All $L^p (\R^{d + 1}_+)$-norms below
are taken with respect to $\dd x \frac{\dd t}{t}$. Whenever $\alpha > 0$ is
not an integer, we write
\[ k_{\alpha} \eqd \lfloor \alpha \rfloor, \qquad \sigma_{\alpha} \eqd \alpha
   - k_{\alpha} \in (0, 1), \qquad n_{\alpha} \eqd k_{\alpha} + 1. \]
A family
\[ \theta = \{\theta_{(x, t)} \}_{(x, t) \in \R^{d + 1}_+} \subset \Sch (\R^d)
\]
is bounded in $\Sch (\R^d)$ if, for every nonnegative integer $M$ and every
multiindex $\beta$,
\[ \sup_{(x, t) \in \R^{d + 1}_+} \sup_{z \in \R^d} \langle z \rangle^M  |
   \partial^{\beta} \theta_{(x, t)} (z) | < \infty . \]
The implicit constants in estimates involving such a family may depend on
finitely many of these uniform Schwartz seminorms.

\begin{definition}[Cancellation classes]
  \label{def:psi-classes}The set $\Psi (\R^d) \subset \Sch (\R^d)$ is the set
  of Schwartz functions whose Fourier transform is supported away from the
  origin:
  \begin{equation}
    \label{eq:psi-fourier-support} \Psi (\R^d) \eqd \left\{ f \in \Sch (\R^d)
    \st 0 \nin \spt \FT{f} \right\} .
  \end{equation}
  Equivalently, $\FT{f}$ vanishes in a neighbourhood of the origin. In
  particular, $\Psi (\R^d) \subset \Psi^m (\R^d)$ for every nonnegative
  integer $m$.
  
  For a nonnegative integer $m$, the set $\Psi^m (\R^d) \subset \Sch (\R^d)$
  is the set of Schwartz functions with vanishing moments up to order $m$:
  \begin{equation}
    \label{eq:psi-vanishing-moments} \Psi^m (\R^d) \eqd \left\{ f \in \Sch
    (\R^d) \st \int_{\R^d} f (x) P (x) \hspace{0.17em} \dd x = 0 \quad
    \text{for every polynomial } P \text{with } \deg (P) \le m \right\} .
  \end{equation}
\end{definition}

\begin{definition}[H{\"o}lder spaces]
  \label{def:holder-spaces}Let $\alpha > 0$ be noninteger. For $f \in
  C^{k_{\alpha}} (\R^d)$, define the homogeneous H{\"o}lder seminorm by
  \begin{equation}
    \label{eq:holder-homogeneous-seminorm} \|f\|_{\dot{\mathcal{C}}^{\alpha}
    (\R^d)} \eqd \max_{| \gamma | = k_{\alpha}} \sup_{x \neq x'}  \frac{|
    \partial^{\gamma} f (x') - \partial^{\gamma} f (x) |}{|x' -
    x|^{\sigma_{\alpha}}} .
  \end{equation}
  The homogeneous space is used modulo the kernel of this seminorm, namely
  modulo polynomials of degree at most $k_{\alpha}$.
  
  Define the nonhomogeneous H{\"o}lder norm by
  \begin{equation}
    \label{eq:holder-nonhomogeneous-norm} \|f\|_{\mathcal{C}^{\alpha} (\R^d)}
    \eqd \sum_{| \gamma | \le k_{\alpha}} \| \partial^{\gamma}
    f\|_{L^{\infty}} + \max_{| \gamma | = k_{\alpha}}
    \sup_{\tmscript{\begin{array}{c}
      x \neq x'\\
      |x' - x| < 1
    \end{array}}}  \frac{| \partial^{\gamma} f (x') - \partial^{\gamma} f (x)
    |}{|x' - x|^{\sigma_{\alpha}}} .
  \end{equation}
  We also use the finite-difference notation
  \[ \Delta_h f (y) \eqd f (y + h) - f (y), \qquad \Delta_h^n \eqd \Delta_h
     \circ \Delta_h^{n - 1}  \quad (n \ge 2) . \]
  For positive noninteger $\alpha$, the standard finite-difference
  characterization gives
  \begin{equation}
    \label{eq:holder-homogeneous-finite-difference}
    \|f\|_{\dot{\mathcal{C}}^{\alpha} (\R^d)} \simeq_{d, \alpha}
    \sup_{\tmscript{\begin{array}{c}
      y \in \R^d\\
      h \in \R^d \setminus \{0\}
    \end{array}}}  \frac{| \Delta_h^{n_{\alpha}} f (y) |}{|h|^{\alpha}}
  \end{equation}
  on the quotient by polynomials of degree at most $k_{\alpha}$, and
  \begin{equation}
    \label{eq:holder-nonhomogeneous-finite-difference}
    \|f\|_{\mathcal{C}^{\alpha} (\R^d)} \simeq_{d, \alpha} \|f\|_{L^{\infty}}
    + \sup_{\tmscript{\begin{array}{c}
      y \in \R^d\\
      0 < |h| < 1
    \end{array}}}  \frac{| \Delta_h^{n_{\alpha}} f (y) |}{|h|^{\alpha}} .
  \end{equation}
\end{definition}

\begin{definition}[Wavelet coefficients]
  \label{def:wavelet-coefficients}For $\varphi \in \Sch (\R^d)$ and $(x, t)
  \in \R^{d + 1}_+$, define
  \[ \Tr_x \varphi (y) \eqd \varphi (y - x), \qquad \Dil_t \varphi (y) \eqd
     t^{- d} \varphi (y / t) . \]
  For a function $f$ of at most polynomial growth and $\varphi \in \Sch
  (\R^d)$, define
  \begin{equation}
    \label{eq:wavelet-coefficient-fixed-test} \Em_{\varphi} [f] (x, t) \eqd
    \int_{\R^d} f (y) \Tr_x \Dil_t \varphi (y) \hspace{0.17em} \dd y
  \end{equation}
  whenever the integral is absolutely convergent.
\end{definition}

\begin{remark}
  \label{rmk:variable-wavelets}We also allow the test function to depend on
  $(x, t)$. Thus, if
  \[ \varphi : (x, t) \mapsto \varphi_{(x, t)} \in \Sch (\R^d), \]
  then
  \begin{equation}
    \label{eq:wavelet-coefficient-family} \Em_{\varphi} [f] (x, t) \eqd
    \int_{\R^d} f (y) \Tr_x \Dil_t \varphi_{(x, t)} (y) \hspace{0.17em} \dd y
  \end{equation}
  whenever the integral is absolutely convergent.
\end{remark}

\begin{proposition}[Wavelet coefficient bounds for H{\"o}lder spaces]
  \label{prop:holder-analysis}Let $\alpha > 0$ be noninteger. Let
  \[ \psi : \R^{d + 1}_+ \to \Psi^{k_{\alpha}} (\R^d) \]
  be a bounded family. Then, for every $f \in C^{k_{\alpha}} (\R^d)$ with
  $\|f\|_{\dot{\mathcal{C}}^{\alpha} (\R^d)} < \infty$, it holds that
  \begin{equation}
    \label{eq:holder-analysis-homogeneous} \left| t^{- \alpha} \Em_{\psi} [f]
    \right|_{L^{\infty} (\R^{d + 1}_+)} \lesssim_{d, \alpha, \psi}
    \|f\|_{\dot{\mathcal{C}}^{\alpha} (\R^d)} .
  \end{equation}
  Furthermore, if
  \[ \varphi : \R^{d + 1}_+ \to \Sch (\R^d) \]
  is a bounded family, then, for every $f \in \mathcal{C}^{\alpha} (\R^d)$,
  \begin{equation}
    \label{eq:holder-analysis-nonhomogeneous} \left| \1_{t \approx 1} t^{-
    \alpha} \Em_{\varphi} [f] \right|_{L^{\infty} (\R^{d + 1}_+)} + \left|
    \1_{t < 1} t^{- \alpha} \Em_{\psi} [f] \right|_{L^{\infty} (\R^{d + 1}_+)}
    \lesssim_{d, \alpha, \psi, \varphi, c_0, C_0} \|f\|_{\mathcal{C}^{\alpha}
    (\R^d)} .
  \end{equation}
\end{proposition}

Let $k = k_{\alpha}$. We first prove \eqref{eq:holder-analysis-homogeneous}.
For fixed $x \in \R^d$, define the Taylor polynomial
\[ P_{x, k}  (x + tz) \eqd \sum_{| \gamma | \le k} \frac{t^{| \gamma |}
   z^{\gamma}}{\gamma !} \partial^{\gamma} f (x) . \]
Since $\psi_{(x, t)} \in \Psi^k (\R^d)$, all moments of $\psi_{(x, t)}$ of
order at most $k$ vanish. Hence
\[ \Em_{\psi} [f] (x, t) = \int_{\R^d} (f (x + tz) - P_{x, k} (x + tz))
   \psi_{(x, t)} (z) \hspace{0.17em} \dd z. \]
Taylor's theorem with H{\"o}lder remainder gives
\[ |f (x + tz) - P_{x, k} (x + tz) | \lesssim_{d, \alpha}
   \|f\|_{\dot{\mathcal{C}}^{\alpha} (\R^d)} t^{\alpha} |z|^{\alpha} . \]
Therefore
\[ | \Em_{\psi} [f] (x, t) | \begin{aligned}{t}
     & \le \|f\|_{\dot{\mathcal{C}}^{\alpha} (\R^d)} t^{\alpha}  \int_{\R^d}
     |z|^{\alpha} | \psi_{(x, t)} (z) | \hspace{0.17em} \dd z\\
     & \lesssim_{d, \alpha, \psi} t^{\alpha}
     \|f\|_{\dot{\mathcal{C}}^{\alpha} (\R^d)} .
   \end{aligned} \]
This proves \eqref{eq:holder-analysis-homogeneous}.

Next assume $f \in \mathcal{C}^{\alpha} (\R^d)$ and $0 < t < 1$. The same
Taylor polynomial satisfies
\[ |f (x + tz) - P_{x, k} (x + tz) | \lesssim_{d, \alpha}
   \|f\|_{\mathcal{C}^{\alpha} (\R^d)} t^{\alpha} \langle z \rangle^{\alpha} .
\]
Indeed, this is the usual Taylor remainder estimate when $|tz| < 1$. When
$|tz| \ge 1$, the $L^{\infty}$-bounds on $\partial^{\gamma} f$, $| \gamma |
\le k$, give the same estimate because $k < \alpha$. Using the vanishing
moments of $\psi_{(x, t)}$, we obtain
\[ | \Em_{\psi} [f] (x, t) | \lesssim_{d, \alpha, \psi} t^{\alpha}
   \|f\|_{\mathcal{C}^{\alpha} (\R^d)}  \qquad (0 < t < 1) . \]
This proves the second term in \eqref{eq:holder-analysis-nonhomogeneous}.

It remains to estimate the term with $t \approx 1$. No cancellation is needed.
Since $\varphi$ is bounded in the Schwartz topology,
\[ | \Em_{\varphi} [f] (x, t) | \le \|f\|_{L^{\infty}}  \int_{\R^d} |
   \varphi_{(x, t)} (z) | \hspace{0.17em} \dd z \lesssim_{\varphi}
   \|f\|_{\mathcal{C}^{\alpha} (\R^d)} . \]
On the support of $\1_{t \approx 1}$, the factor $t^{- \alpha}$ is bounded by
a constant depending only on $\alpha, c_0, C_0$. This proves
\eqref{eq:holder-analysis-nonhomogeneous}.

\begin{definition}[Reconstruction operator]
  \label{def:reconstruction-operator}Let $F : \R^{d + 1}_+ \to \C$, and let
  \[ \varphi : \R^{d + 1}_+ \to \Sch (\R^d) \]
  be a bounded family of test functions. Whenever the integral is absolutely
  convergent, define
  \begin{equation}
    \label{eq:reconstruction-operator} (\mathcal{R}_{\varphi} [F]) (x) \eqd
    \int_0^{\infty} \int_{\R^d} F (y, t) \Tr_y \Dil_t \varphi_{(y, t)} (x)
    \hspace{0.17em} \dd y \frac{\dd t}{t} .
  \end{equation}
  If $E \subset (0, \infty)$, then $\mathcal{R}_{\varphi} [\1_E F]$ denotes
  the same operator with $F (x, t)$ replaced by $\1_E (t) F (x, t)$.
\end{definition}

\begin{lemma}[Kernel bounds]
  \label{lem:kernel-bounds}Let
  \[ \theta : \R^{d + 1}_+ \to \Sch (\R^d) \]
  be a bounded family. Then, for every multiindex $\gamma$, every integer $n
  \ge 1$, every $t > 0$, and every $y, h \in \R^d$, it holds that
  \begin{equation}
    \label{eq:kernel-derivative-l1} \int_{\R^d} \left| \partial_y^{\gamma}
    \Tr_x \Dil_t \theta_{(x, t)} (y) \right| \hspace{0.17em} \dd x
    \lesssim_{d, \gamma, \theta} t^{- | \gamma |}
  \end{equation}
  and
  \begin{equation}
    \label{eq:kernel-finite-difference-bound} \int_{\R^d} \left| \Delta_h^n
    \partial_y^{\gamma} \Tr_x \Dil_t \theta_{(x, t)} (y) \right|
    \hspace{0.17em} \dd x \lesssim_{d, \gamma, n, \theta} t^{- | \gamma |}
    \min \left\{ 1, \left( \frac{|h|}{t} \right)^n \right\} .
  \end{equation}
  In particular, taking $\gamma = 0$ and $n = 1$ gives
  \[ \int_{\R^d} \left| \Tr_x \Dil_t \theta_{(x, t)} (y + h) - \Tr_x \Dil_t
     \theta_{(x, t)} (y) \right| \hspace{0.17em} \dd x \lesssim_{d, \theta}
     \min \left\{ 1, \frac{|h|}{t} \right\} . \]
\end{lemma}

{\fromLLM{I adjusted the proof of  Lemma~\ref{lem:kernel-bounds}: the
finite-difference derivative estimate is proved on the range $|h| \le t$, and
the range $|h| > t$ is handled by the crude $L^1$ estimate. This avoids losing
a factor from the length of the segment in the mean-value representation.}}

Choose an integer $N > d$. By boundedness in the Schwartz topology,
\[ \left| \partial_y^{\gamma} \Tr_x \Dil_t \theta_{(x, t)} (y) \right|
   \lesssim_{\gamma, \theta, N} t^{- d - | \gamma |} \left\langle \frac{y -
   x}{t} \right\rangle^{- N} . \]
Integrating in $x$ gives \eqref{eq:kernel-derivative-l1}.

The estimate
\[ \int_{\R^d} \left| \Delta_h^n \partial_y^{\gamma} \Tr_x \Dil_t \theta_{(x,
   t)} (y) \right| \hspace{0.17em} \dd x \lesssim_{d, \gamma, n, \theta} t^{-
   | \gamma |} \]
follows from \eqref{eq:kernel-derivative-l1} and the identity
\[ \Delta_h^n g = \sum_{j = 0}^n (- 1)^{n - j} \binom{n}{j} g (\cdummy + jh) .
\]
It remains to obtain the estimate with $(|h| / t)^n$. If $|h| > t$, the
preceding bound already implies this estimate. Hence assume $|h| \le t$. The
iterated mean value theorem gives
\[ \Delta_h^n g (y) = \int_{[0, 1]^n} D_h^n g (y + (s_1 + \cdots + s_n) h)
   \hspace{0.17em} \dd s_1 \cdots \dd s_n . \]
Applying this to
\[ g (y) = \partial_y^{\gamma} \Tr_x \Dil_t \theta_{(x, t)} (y), \]
and using the uniform Schwartz bounds for derivatives of order $| \gamma | +
n$, gives
\[ \left| \Delta_h^n \partial_y^{\gamma} \Tr_x \Dil_t \theta_{(x, t)} (y)
   \right| \lesssim_{\gamma, n, \theta, N} |h|^n t^{- d - | \gamma | - n}
   \sup_{0 \le s \le n} \left\langle \frac{y + sh - x}{t} \right\rangle^{- N}
   . \]
Since $|h| \le t$, the last supremum is bounded by
\[ \lesssim_{n, N} \left\langle \frac{y - x}{t} \right\rangle^{- N} . \]
Integrating in $x$ yields
\[ \int_{\R^d} \left| \Delta_h^n \partial_y^{\gamma} \Tr_x \Dil_t \theta_{(x,
   t)} (y) \right| \hspace{0.17em} \dd x \lesssim_{d, \gamma, n, \theta} t^{-
   | \gamma |} \left( \frac{|h|}{t} \right)^n . \]
Combining this estimate with the crude $L^1$-estimate proves
\eqref{eq:kernel-finite-difference-bound}.

\begin{proposition}[Reconstruction of H{\"o}lder spaces]
  \label{prop:holder-reconstruction}Let $\alpha > 0$ be noninteger. Let $F :
  \R^{d + 1}_+ \to \C$ be measurable and assume
  \begin{equation}
    \label{eq:holder-reconstruction-size-assumption} A \eqd | t^{- \alpha} F
    |_{L^{\infty} (\R^{d + 1}_+)} < \infty .
  \end{equation}
  Let
  \[ \psi : \R^{d + 1}_+ \to \Psi^{k_{\alpha}} (\R^d) \]
  be a bounded family. The homogeneous reconstruction $\mathcal{R}_{\psi} [F]$
  is the class modulo polynomials of degree at most $k_{\alpha}$ determined by
  the absolutely convergent finite-difference formula
  \begin{equation}
    \label{eq:homogeneous-reconstruction-increment} \Delta_h^{n_{\alpha}}
    (\mathcal{R}_{\psi} [F]) (y) \eqd \int_0^{\infty} \int_{\R^d} F (x, t)
    \Delta_h^{n_{\alpha}} \left[ \Tr_x \Dil_t \psi_{(x, t)} \right] (y)
    \hspace{0.17em} \dd x \frac{\dd t}{t} .
  \end{equation}
  It satisfies
  \begin{equation}
    \label{eq:holder-reconstruction-homogeneous} \|\mathcal{R}_{\psi}
    [F]\|_{\dot{\mathcal{C}}^{\alpha} (\R^d)} \lesssim_{d, \alpha, \psi} |
    t^{- \alpha} F |_{L^{\infty} (\R^{d + 1}_+)} .
  \end{equation}
  Furthermore, if
  \[ \varphi : \R^{d + 1}_+ \to \Sch (\R^d) \]
  is a bounded family, then the expression
  \[ \mathcal{R}_{\psi} [\1_{t \le 1} F] +\mathcal{R}_{\varphi} [\1_{t \approx
     1} F] \]
  is absolutely convergent pointwise and satisfies
  \begin{equation}
    \label{eq:holder-reconstruction-nonhomogeneous} \left| \mathcal{R}_{\psi}
    [\1_{t \le 1} F] +\mathcal{R}_{\varphi} [\1_{t \approx 1} F]
    \right|_{\mathcal{C}^{\alpha} (\R^d)} \lesssim_{d, \alpha, \psi, \varphi,
    c_0, C_0} | t^{- \alpha} F |_{L^{\infty} (\R^{d + 1}_+)} .
  \end{equation}
\end{proposition}

\begin{remark}[Representative in the homogeneous reconstruction]
  \label{rmk:homogeneous-reconstruction-representative}The homogeneous
  reconstruction in  Proposition~\ref{prop:holder-reconstruction} is defined
  modulo polynomials of degree at most $k_{\alpha}$. If $\alpha \in (0, 1)$,
  this is only a constant ambiguity, and the convention $f (0) = 0$ selects a
  representative. If $k_{\alpha} \ge 1$, the same convention fixes only the
  constant coefficient. A full normalization may be imposed by requiring
  \[ \partial^{\gamma} f (0) = 0 \qquad \text{for every } | \gamma | \le
     k_{\alpha} . \]
  The estimates above are invariant under the choice of representative.
\end{remark}

{\step{Homogeneous reconstruction.}} Let $n = n_{\alpha}$. By 
Lemma~\ref{lem:kernel-bounds},
\[ \int_0^{\infty} \int_{\R^d} |F (x, t) | \left| \Delta_h^n \left[ \Tr_x
   \Dil_t \psi_{(x, t)} \right] (y) \right| \hspace{0.17em} \dd x \frac{\dd
   t}{t} \begin{aligned}{t}
     & \lesssim_{d, n, \psi} A \int_0^{\infty} t^{\alpha} \min \left\{ 1,
     \left( \frac{|h|}{t} \right)^n \right\} \frac{\dd t}{t}\\
     & \lesssim_{\alpha, n, d, \psi} A |h|^{\alpha} .
   \end{aligned} \]
The last integral is finite because $\alpha > 0$ and $n > \alpha$. More
explicitly, for $h \neq 0$,
\[ \int_0^{\infty} t^{\alpha} \min \left\{ 1, \left( \frac{|h|}{t} \right)^n
   \right\} \frac{\dd t}{t} = |h|^{\alpha}  \left( \int_0^1 s^{\alpha} 
   \frac{\dd s}{s} + \int_1^{\infty} s^{\alpha - n}  \frac{\dd s}{s} \right) .
\]
Thus the integral in \eqref{eq:homogeneous-reconstruction-increment} is
absolutely convergent and satisfies
\[ | \Delta_h^n (\mathcal{R}_{\psi} [F]) (y) | \lesssim_{d, \alpha, \psi} A
   |h|^{\alpha} . \]
The formula is consistent with finite differences of smooth truncated
syntheses. Indeed, for $0 < \varepsilon < R < \infty$, set
\[ G_{\varepsilon, R} (y) \eqd \int_{\varepsilon}^R \int_{\R^d} F (x, t) \Tr_x
   \Dil_t \psi_{(x, t)} (y) \hspace{0.17em} \dd x \frac{\dd t}{t} . \]
The preceding tail estimates show that $\Delta_h^n G_{\varepsilon, R} (y)$
converges as $\varepsilon \to 0^+$ and $R \to \infty$, for every $y, h \in
\R^d$. Hence the limit determines an element of the homogeneous quotient
modulo polynomials of degree at most $n - 1 = k_{\alpha}$. By the
finite-difference characterization
\eqref{eq:holder-homogeneous-finite-difference}, this gives
\eqref{eq:holder-reconstruction-homogeneous}.

\begin{definition}[Homogeneous smooth convolution singular integrals]
  \label{def:smooth-convolution-singular-integrals}Let $s \in \R$. A
  multiplier $m \in C^{\infty}  (\R^d \setminus \{0\})$ is a homogeneous
  symbol of Fourier order $s$ if, for every nonnegative integer $M$,
  \begin{equation}
    \label{eq:symbol-seminorm} \mathfrak{S}_{s, M} (m) \eqd \max_{| \mu | \le
    M} \sup_{\xi \in \R^d \setminus \{0\}} | \xi |^{- s + | \mu |}  |
    \partial^{\mu} m (\xi) | < \infty .
  \end{equation}
  For $g \in \Psi (\R^d)$, define
  \[ T_m g \eqd \mathcal{F}^{- 1} (m \FT{g}) . \]
  This is a Schwartz function, since $\FT{g}$ vanishes in a neighbourhood of
  the origin. We call $T_m$ the corresponding smooth convolution singular
  integral operator of Fourier order $s$.
\end{definition}

\begin{remark}[Low-frequency convention for homogeneous symbols]
  \label{rmk:homogeneous-symbol-caveat}The definition above does not choose an
  extension of $m$ across $\xi = 0$. If two tempered extensions are chosen,
  their difference is a distribution supported at $\{0\}$, and therefore the
  inverse Fourier transform of the difference is a polynomial. Thus the
  homogeneous action on H{\"o}lder classes is well defined modulo the
  polynomial ambiguity already present in $\dot{\mathcal{C}}^{\alpha}$. A
  nonhomogeneous estimate on $\mathcal{C}^{\alpha}$ would require an
  additional low-frequency normalization or cutoff.
\end{remark}

\begin{lemma}[Continuous Calder{\'o}n reproducing formula]
  \label{lem:continuous-calderon}There exist $\phi, \psi \in \Psi (\R^d)$ such
  that
  \begin{equation}
    \label{eq:calderon-partition} \int_0^{\infty} \FT{\psi} (- t \xi)
    \FT{\phi} (t \xi) \frac{\dd t}{t} = 1 \qquad \text{for every } \xi \in
    \R^d \setminus \{0\} .
  \end{equation}
  For such a pair, regarded as constant families in $(x, t)$, it holds that
  \begin{equation}
    \label{eq:calderon-reconstruction} f =\mathcal{R}_{\phi} [\Em_{\psi} [f]]
  \end{equation}
  in $\Sch' (\R^d)$ for every $f \in \Sch (\R^d)$. More generally, for every
  positive noninteger $\alpha$, the same identity holds in the homogeneous
  quotient for $f \in \dot{\mathcal{C}}^{\alpha} (\R^d)$.
\end{lemma}

Choose $\rho \in C_c^{\infty} ((1 / 2, 2))$ so that
\[ \int_0^{\infty} \rho (r)^2 \frac{\dd r}{r} = 1. \]
After the harmless normalization dictated by the Fourier transform convention,
define $\phi$ and $\psi$ by
\[ \FT{\phi} (\xi) = \rho (| \xi |), \qquad \FT{\psi} (\xi) = \rho (| \xi |) .
\]
Then $\phi, \psi \in \Psi (\R^d)$, and \eqref{eq:calderon-partition} follows
by the change of variables $r = t | \xi |$.

Taking the Fourier transform of $\mathcal{R}_{\phi} [\Em_{\psi} [f]]$ gives
the multiplier
\[ \xi \mapsto \int_0^{\infty} \FT{\psi} (- t \xi) \FT{\phi} (t \xi) \frac{\dd
   t}{t}, \]
which is equal to $1$ for $\xi \neq 0$. Hence
\eqref{eq:calderon-reconstruction} holds in $\Sch' (\R^d)$. The extension to
homogeneous H{\"o}lder classes follows by testing against functions in $\Psi
(\R^d)$; polynomials are precisely the ambiguity invisible to these tests.

\begin{lemma}[Action of homogeneous symbols on wavelets]
  \label{lem:symbol-action-wavelets}Let $s \in \R$, and let $m$ be a
  homogeneous symbol of Fourier order $s$ in the sense of 
  Definition~\ref{def:smooth-convolution-singular-integrals}. Let $\phi \in
  \Psi (\R^d)$. For $t > 0$, define $\theta_t \in \Sch (\R^d)$ by
  \begin{equation}
    \label{eq:symbol-action-wavelet} \FT{\theta_t} (\eta) \eqd t^s m (t^{- 1}
    \eta) \FT{\phi} (\eta), \qquad \eta \in \R^d .
  \end{equation}
  Then $\theta_t \in \Psi (\R^d)$, and the family
  \[ (x, t) \mapsto \theta_{(x, t)} \eqd \theta_t \]
  is bounded in $\Sch (\R^d)$. Moreover,
  \begin{equation}
    \label{eq:symbol-action-wavelet-identity} T_m (\Tr_x \Dil_t \phi) = t^{-
    s} \Tr_x \Dil_t \theta_t
  \end{equation}
  as Schwartz functions, for every $x \in \R^d$ and $t > 0$.
\end{lemma}

Since $\FT{\phi}$ vanishes in a neighbourhood of the origin, the same is true
of $\FT{\theta_t}$. Hence $\theta_t \in \Psi (\R^d)$.

We next prove uniform Schwartz bounds. For each multiindex $\nu$,
\[ \partial_{\eta}^{\nu}  [t^s m (t^{- 1} \eta)] = t^{s - | \nu |} 
   (\partial^{\nu} m)  (t^{- 1} \eta), \]
and therefore
\[ | \partial_{\eta}^{\nu}  [t^s m (t^{- 1} \eta)] | \lesssim_{\nu}
   \mathfrak{S}_{s, | \nu |} (m) | \eta |^{s - | \nu |}  \qquad (\eta \neq 0)
   . \]
Multiplication by $\FT{\phi}$, which vanishes near the origin and is rapidly
decreasing at infinity, gives uniform Schwartz bounds for $\FT{\theta_t}$.
Fourier inversion gives the corresponding uniform Schwartz bounds for
$\theta_t$.

Finally, taking Fourier transforms in the spatial variable gives
\[ \FT{T_m (\Tr_x \Dil_t \phi)} (\xi) = m (\xi) e^{- ix \cdot \xi} \FT{\phi}
   (t \xi), \]
while
\[ \FT{t^{- s} \Tr_x \Dil_t \theta_t} (\xi) = t^{- s} e^{- ix \cdot \xi}
   \FT{\theta_t} (t \xi) = m (\xi) e^{- ix \cdot \xi} \FT{\phi} (t \xi) . \]
This proves \eqref{eq:symbol-action-wavelet-identity}.

\begin{proposition}[Homogeneous H{\"o}lder boundedness of smooth convolution
singular integrals]
  \label{prop:smooth-convolution-singular-integral-holder}Let $s \in \R$, and
  let $m$ be a homogeneous symbol of Fourier order $s$. Let $\alpha > 0$ be
  noninteger and set
  \[ \beta \eqd \alpha - s. \]
  Assume that $\beta > 0$ and that $\beta$ is not an integer. Fix a
  Calder{\'o}n pair $(\phi, \psi)$ satisfying \eqref{eq:calderon-partition}.
  Then $T_m$ extends to a bounded operator
  \[ T_m : \dot{\mathcal{C}}^{\alpha} (\R^d) \to \dot{\mathcal{C}}^{\beta}
     (\R^d), \]
  modulo the polynomial ambiguities in the homogeneous spaces. More precisely,
  there exists an integer $M_0$, depending only on $d$ and $\beta$, such that
  \begin{equation}
    \label{eq:smooth-singular-integral-holder-bound} \|T_m
    f\|_{\dot{\mathcal{C}}^{\beta} (\R^d)} \lesssim_{d, \alpha, s, \phi, \psi}
    \mathfrak{S}_{s, M_0} (m) \|f\|_{\dot{\mathcal{C}}^{\alpha} (\R^d)} .
  \end{equation}
\end{proposition}

Choose a representative of $f$. Since $\psi \in \Psi (\R^d)$, the coefficient
\[ F (x, t) \eqd \Em_{\psi} [f] (x, t) \]
is independent of the representative of the homogeneous class. By 
Proposition~\ref{prop:holder-analysis},
\begin{equation}
  \label{eq:singular-integral-analysis-size} | t^{- \alpha} F |_{L^{\infty}
  (\R^{d + 1}_+)} \lesssim_{d, \alpha, \psi} \|f\|_{\dot{\mathcal{C}}^{\alpha}
  (\R^d)} .
\end{equation}
Let $\theta_t$ be defined from $m$ and $\phi$ by
\eqref{eq:symbol-action-wavelet}, and write $\theta_{(x, t)} \eqd \theta_t$.
By  Lemma~\ref{lem:symbol-action-wavelets}, this is a bounded family in $\Psi
(\R^d)$, hence in $\Psi^{k_{\beta}} (\R^d)$. Moreover, the finite collection
of Schwartz seminorms of $\theta$ used in 
Proposition~\ref{prop:holder-reconstruction} is bounded by
\[ \lesssim_{d, \alpha, s, \phi} \mathfrak{S}_{s, M_0} (m) \]
for some integer $M_0 = M_0 (d, \beta)$.

Set
\[ G (x, t) \eqd t^{- s} F (x, t) . \]
Since $\beta = \alpha - s$,
\[ | t^{- \beta} G |_{L^{\infty} (\R^{d + 1}_+)} = | t^{- \alpha} F
   |_{L^{\infty} (\R^{d + 1}_+)} . \]
The homogeneous reconstruction estimate
\eqref{eq:holder-reconstruction-homogeneous} therefore gives
\[ \|\mathcal{R}_{\theta} [G]\|_{\dot{\mathcal{C}}^{\beta} (\R^d)}
   \lesssim_{d, \alpha, s, \phi, \psi} \mathfrak{S}_{s, M_0} (m)
   \|f\|_{\dot{\mathcal{C}}^{\alpha} (\R^d)} . \]
We define $T_m f$ in the homogeneous quotient by
\begin{equation}
  \label{eq:smooth-singular-integral-extension} T_m f \eqd
  \mathcal{R}_{\theta} \left[ t^{- s} \Em_{\psi} [f] \right] .
\end{equation}
This definition gives the bound
\eqref{eq:smooth-singular-integral-holder-bound}.

It remains to check that this extension agrees with the Fourier multiplier on
test functions. Let $f \in \Sch (\R^d)$. Taking Fourier transforms in the
reconstruction formula and using \eqref{eq:symbol-action-wavelet} gives, for
$\xi \neq 0$,
\[ \FT{\mathcal{R}_{\theta} \left[ t^{- s} \Em_{\psi} [f] \right]} (\xi) = m
   (\xi) \FT{f} (\xi)  \int_0^{\infty} \FT{\psi} (- t \xi) \FT{\phi} (t \xi)
   \frac{\dd t}{t} . \]
By \eqref{eq:calderon-partition}, the integral is equal to $1$. Thus the
extension agrees with the multiplier $m$ away from $\xi = 0$. Any difference
caused by a choice of extension of $m$ at $\xi = 0$ is a polynomial in
physical space, and is therefore invisible in the homogeneous target space.
This concludes the proof.

\begin{corollary}[Smoothing sign convention]

  noninteger, and assume that $\alpha + s > 0$ is not an integer. If $m$
  satisfies

  \begin{equation}
    \label{eq:smoothing-sign-convention-bound} \|T_m
    f\|_{\dot{\mathcal{C}}^{\alpha + s} (\R^d)} \lesssim_{d, \alpha, s, \phi,
    \psi} \mathfrak{S}_{- s, M_0} (m) \|f\|_{\dot{\mathcal{C}}^{\alpha}
    (\R^d)} .
  \end{equation}
\end{corollary}
